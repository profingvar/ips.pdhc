# IPS Server — Initial SQL Design (Reference Only)
#
# This file is a suggested database design (Rule 5).
# All tables use UUID/GUID primary keys (Rule 18).
# PostgreSQL with JSONB resource storage (description_of_IPS.md).
# FHIR R5 compliant naming and structure (Rule 15).
#
# Status: DRAFT — awaiting review before implementation.

# ============================================================
# 1. CORE IDENTITY
# ============================================================

CREATE TABLE users (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username        VARCHAR(255) NOT NULL UNIQUE,
    display_name    VARCHAR(255),
    email           VARCHAR(255),
    password_hash   VARCHAR(255),          -- for bootstrap SU only; normal auth is SSO
    role            VARCHAR(50) NOT NULL DEFAULT 'operator',
                                           -- e.g. admin, operator, service_account
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    is_superuser    BOOLEAN NOT NULL DEFAULT FALSE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE clinics (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name            VARCHAR(255) NOT NULL,
    identifier      VARCHAR(255) UNIQUE,   -- org-level code or OID
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE user_clinic_assignments (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_guid       UUID NOT NULL REFERENCES users(guid) ON DELETE CASCADE,
    clinic_guid     UUID NOT NULL REFERENCES clinics(guid) ON DELETE CASCADE,
    role            VARCHAR(50) NOT NULL DEFAULT 'member',
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (user_guid, clinic_guid)
);

# ============================================================
# 2. API KEY MANAGEMENT (Rule 8)
# ============================================================

CREATE TABLE api_keys (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_guid       UUID REFERENCES users(guid) ON DELETE SET NULL,
    key_hash        VARCHAR(255) NOT NULL UNIQUE,   -- never store plaintext
    label           VARCHAR(255),                    -- human-friendly name
    prefix          VARCHAR(12) NOT NULL,            -- visible prefix for identification
    scopes          JSONB DEFAULT '[]',              -- e.g. ["read:fhir","write:fhir"]
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    expires_at      TIMESTAMPTZ,                     -- NULL = no expiry
    revoked_at      TIMESTAMPTZ,                     -- set on revocation
    last_used_at    TIMESTAMPTZ,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
# Rules: store hashed only; rotate periodically; revoke sets revoked_at
#        + is_active=false; expired keys are rejected at validation time.

# ============================================================
# 3. FHIR RESOURCE STORAGE
# ============================================================

# Generic resource table — stores any FHIR R5 resource as JSONB.
# This is the "store the canonical JSON" approach described in
# description_of_IPS.md (Phase 2).

CREATE TABLE fhir_resources (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    resource_type   VARCHAR(64) NOT NULL,          -- Patient, Condition, Observation, etc.
    resource_id     VARCHAR(255) NOT NULL,          -- FHIR logical id (also a UUID)
    version_id      INTEGER NOT NULL DEFAULT 1,
    resource_json   JSONB NOT NULL,                 -- full FHIR resource as received
    patient_guid    UUID,                           -- FK for patient-scoped resources
    status          VARCHAR(32) DEFAULT 'active',
    last_updated    TIMESTAMPTZ NOT NULL DEFAULT now(),
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (resource_type, resource_id, version_id)
);

# Indexes for common FHIR search patterns
CREATE INDEX idx_fhir_resources_type ON fhir_resources(resource_type);
CREATE INDEX idx_fhir_resources_patient ON fhir_resources(patient_guid);
CREATE INDEX idx_fhir_resources_type_patient ON fhir_resources(resource_type, patient_guid);
CREATE INDEX idx_fhir_resources_last_updated ON fhir_resources(last_updated);
CREATE INDEX idx_fhir_resources_json_gin ON fhir_resources USING GIN (resource_json);

# ============================================================
# 4. PATIENT INDEX
# ============================================================

# Lightweight index table for fast patient lookup.
# The full Patient resource lives in fhir_resources; this table
# caches searchable fields to avoid JSONB queries for common ops.

CREATE TABLE patient_index (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    fhir_resource_guid UUID NOT NULL REFERENCES fhir_resources(guid) ON DELETE CASCADE,
    resource_id     VARCHAR(255) NOT NULL UNIQUE,  -- same as fhir_resources.resource_id
    identifier_system VARCHAR(255),                -- e.g. "urn:oid:1.2.752.129.2.1.3.1" (Swedish PNR)
    identifier_value  VARCHAR(255),                -- the actual identifier value
    family_name     VARCHAR(255),
    given_name      VARCHAR(255),
    birth_date      DATE,
    gender          VARCHAR(20),
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_patient_index_identifier ON patient_index(identifier_system, identifier_value);
CREATE INDEX idx_patient_index_name ON patient_index(family_name, given_name);

CREATE TABLE patient_clinic_assignments (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    patient_guid    UUID NOT NULL REFERENCES patient_index(guid) ON DELETE CASCADE,
    clinic_guid     UUID NOT NULL REFERENCES clinics(guid) ON DELETE CASCADE,
    assigned_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (patient_guid, clinic_guid)
);

# ============================================================
# 5. IPS CARDS (Phase 4)
# ============================================================

# A "card" is a named, owned IPS concept tied to a patient.
# It is the stable handle; snapshots capture point-in-time content.

CREATE TABLE ips_cards (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    patient_guid    UUID NOT NULL REFERENCES patient_index(guid) ON DELETE CASCADE,
    clinic_guid     UUID REFERENCES clinics(guid) ON DELETE SET NULL,
    created_by_guid UUID REFERENCES users(guid) ON DELETE SET NULL,
    title           VARCHAR(255),
    status          VARCHAR(32) NOT NULL DEFAULT 'active',  -- active, archived
    mode            VARCHAR(16) NOT NULL DEFAULT 'full',    -- full, minimal
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_ips_cards_patient ON ips_cards(patient_guid);

# ============================================================
# 6. IPS SNAPSHOTS (Phase 4)
# ============================================================

# Immutable point-in-time capture of a generated IPS bundle.

CREATE TABLE ips_snapshots (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    card_guid       UUID NOT NULL REFERENCES ips_cards(guid) ON DELETE CASCADE,
    bundle_json     JSONB NOT NULL,                -- the full FHIR Bundle
    composition_date TIMESTAMPTZ NOT NULL,          -- "as of" date
    mode            VARCHAR(16) NOT NULL,           -- full, minimal
    generated_by_guid UUID REFERENCES users(guid) ON DELETE SET NULL,
    resource_count  INTEGER,                        -- number of entries in bundle
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
    -- snapshots are immutable: no updated_at
);

CREATE INDEX idx_ips_snapshots_card ON ips_snapshots(card_guid);
CREATE INDEX idx_ips_snapshots_created ON ips_snapshots(created_at);

# ============================================================
# 7. PUSH DESTINATIONS (Phase 5)
# ============================================================

CREATE TABLE push_destinations (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name            VARCHAR(255) NOT NULL,
    destination_type VARCHAR(32) NOT NULL,          -- fhir_endpoint, integration_gateway, webhook
    endpoint_url    VARCHAR(2048) NOT NULL,
    auth_method     VARCHAR(32),                    -- bearer, basic, api_key, none
    auth_config     JSONB DEFAULT '{}',             -- encrypted/hashed credentials reference
    headers         JSONB DEFAULT '{}',             -- custom headers
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

# ============================================================
# 8. PUSH JOBS (Phase 5)
# ============================================================

CREATE TABLE push_jobs (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    snapshot_guid   UUID NOT NULL REFERENCES ips_snapshots(guid) ON DELETE CASCADE,
    destination_guid UUID NOT NULL REFERENCES push_destinations(guid) ON DELETE CASCADE,
    status          VARCHAR(32) NOT NULL DEFAULT 'queued',
                    -- queued, in_progress, completed, failed, cancelled
    attempts        INTEGER NOT NULL DEFAULT 0,
    max_attempts    INTEGER NOT NULL DEFAULT 3,
    last_attempt_at TIMESTAMPTZ,
    completed_at    TIMESTAMPTZ,
    error_message   TEXT,
    response_code   INTEGER,
    initiated_by_guid UUID REFERENCES users(guid) ON DELETE SET NULL,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_push_jobs_status ON push_jobs(status);
CREATE INDEX idx_push_jobs_snapshot ON push_jobs(snapshot_guid);

# ============================================================
# 9. AUDIT LOG (Phase 6)
# ============================================================

# Append-only audit trail. No UPDATE or DELETE should ever run
# against this table in application code.

CREATE TABLE audit_log (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    actor_guid      UUID,                           -- user or api_key owner
    actor_type      VARCHAR(32),                    -- user, api_key, system
    actor_label     VARCHAR(255),                   -- display name at time of event
    patient_guid    UUID,                           -- patient context if applicable
    event_type      VARCHAR(64) NOT NULL,           -- e.g. ips_bundle_read, patient_create,
                                                    --      api_key_rotate, push_job_complete
    resource_type   VARCHAR(64),                    -- FHIR resource type if applicable
    resource_guid   UUID,                           -- target resource GUID
    request_path    VARCHAR(2048),
    request_method  VARCHAR(10),
    ip_address      VARCHAR(45),
    detail          JSONB DEFAULT '{}',             -- additional context
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_audit_log_actor ON audit_log(actor_guid);
CREATE INDEX idx_audit_log_patient ON audit_log(patient_guid);
CREATE INDEX idx_audit_log_event_type ON audit_log(event_type);
CREATE INDEX idx_audit_log_created ON audit_log(created_at);

# ============================================================
# 10. FHIR CAPABILITY METADATA
# ============================================================

# Stores the server's CapabilityStatement resource.
# Updated at startup or when capabilities change.

CREATE TABLE capability_statement (
    guid            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    resource_json   JSONB NOT NULL,
    version         VARCHAR(32) NOT NULL,
    is_current      BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

# ============================================================
# SUMMARY OF TABLES (10 tables + 3 junction/index tables)
# ============================================================
#
#  Identity & access:
#    users, clinics, user_clinic_assignments, api_keys
#
#  FHIR data:
#    fhir_resources, patient_index, patient_clinic_assignments
#
#  IPS workflow:
#    ips_cards, ips_snapshots
#
#  Distribution:
#    push_destinations, push_jobs
#
#  Operations:
#    audit_log, capability_statement
#
# ============================================================
# DESIGN NOTES
# ============================================================
#
# 1. All PKs are UUIDs (Rule 18). No serial/integer IDs exposed.
#
# 2. fhir_resources uses a single-table design with resource_type
#    discriminator. This matches the "store canonical JSON" approach
#    from the project description. Versioning is supported via
#    (resource_type, resource_id, version_id) unique constraint.
#
# 3. patient_index is a denormalised cache of Patient resource
#    fields. It exists to avoid JSONB extraction for frequent
#    lookups (search by identifier, name). It references the
#    canonical Patient in fhir_resources.
#
# 4. Audit log is append-only. The application must never UPDATE
#    or DELETE rows. Retention policies should be enforced at the
#    operational level (e.g. partition by month, archive after N months).
#
# 5. API key secrets are stored as hashes (key_hash). The plaintext
#    is shown to the user exactly once at creation time. The prefix
#    field stores the first few characters for identification in
#    logs and UI without exposing the full key.
#
# 6. push_destinations.auth_config should reference encrypted
#    secrets, not store plaintext credentials. Implementation
#    should use application-level encryption (e.g. Fernet) with
#    key from environment variable.
#
# 7. The GIN index on fhir_resources.resource_json supports
#    JSONB containment queries (@>, ?) for FHIR search parameters
#    that aren't covered by dedicated index columns.
#
# 8. Alembic manages all schema migrations (Rule per description).
#    This file is the starting-point reference; the migration
#    chain in alembic/versions/ is the source of truth.
